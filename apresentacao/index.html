Primeira Tool real

adapters/inbound/tools.py

@mcp.tool(
    name="check-pr-readiness",
    annotations=READ_ONLY_LOCAL_TOOL,
)
def check_pr_readiness(
    title: Annotated[str, Field(description="Título do PR.")],
    changed_files: FilePaths,
    has_tests: bool,
    has_docs: bool = False,
) -> PRReadiness:
    """Avalia se uma mudança está pronta para abrir um PR."""
    return use_cases.check_pr_readiness(
        title, changed_files, has_tests, has_docs
    )

@mcp.resource(
    "repo://engineering-guidelines",
    name="engineering-guidelines",
    mime_type="text/markdown",
)
def engineering_guidelines() -> str:
    """Regras que o DevBrain aplica, com o porquê."""
    return use_cases.get_engineering_guidelines()

@mcp.prompt(
    name="review-feature",
    title="Revisar feature",

def review_feature(
    summary: Annotated[str, Field(description="O que faz.")],
) -> str:
    """Prepara uma revisão com as tools DevBrain."""
    return use_cases.build_review_prompt(summary)
